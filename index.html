<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no" />
  <title>MacroSense AI</title>
  <script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
  <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
  <script src="https://cdn.tailwindcss.com"></script>
  <style>
    body { background-color: #0d0f12; color: #f3f4f6; font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Helvetica, Arial, sans-serif; }
    .neon-text { color: #10b981; }
    .neon-bg { background-color: #10b981; }
    .neon-border { border-color: #10b981; }
    .card-dark { background-color: #16191e; border: 1px solid #23272f; }
  </style>
</head>
<body>
  <div id="root"></div>

  <script type="text/babel">
    const { useState, useEffect } = React;

    function App() {
      const [activeTab, setActiveTab] = useState('dashboard');
      const [geminiApiKey, setGeminiApiKey] = useState('');
      const [keySavedMessage, setKeySavedMessage] = useState(false);

      // Goals & Macro State
      const [caloriesGoal, setCaloriesGoal] = useState(2200);
      const [proteinGoal, setProteinGoal] = useState(160);
      const [carbsGoal, setCarbsGoal] = useState(220);
      const [fatGoal, setFatGoal] = useState(73);

      // Load saved API key and goals on startup
      useEffect(() => {
        const savedKey = localStorage.getItem('MS_GEMINI_API_KEY');
        if (savedKey) setGeminiApiKey(savedKey);

        const savedGoals = localStorage.getItem('MS_USER_GOALS');
        if (savedGoals) {
          const parsed = JSON.parse(savedGoals);
          if (parsed.calories) setCaloriesGoal(parsed.calories);
          if (parsed.protein) setProteinGoal(parsed.protein);
          if (parsed.carbs) setCarbsGoal(parsed.carbs);
          if (parsed.fat) setFatGoal(parsed.fat);
        }
      }, []);

      // Auto-Balancer for Macros
      const handleProteinChange = (val) => {
        const p = Number(val) || 0;
        setProteinGoal(p);
        const pCals = p * 4;
        const remCals = Math.max(0, caloriesGoal - pCals);
        // Split 60/40 between carbs and fats
        setCarbsGoal(Math.round((remCals * 0.6) / 4));
        setFatGoal(Math.round((remCals * 0.4) / 9));
      };

      // Save API Key & Goals to LocalStorage
      const saveSettings = () => {
        localStorage.setItem('MS_GEMINI_API_KEY', geminiApiKey);
        localStorage.setItem('MS_USER_GOALS', JSON.stringify({
          calories: caloriesGoal,
          protein: proteinGoal,
          carbs: carbsGoal,
          fat: fatGoal
        }));
        setKeySavedMessage(true);
        setTimeout(() => setKeySavedMessage(false), 3000);
      };

      return (
        <div className="max-w-md mx-auto min-h-screen pb-20 px-4 pt-6">
          {/* Header */}
          <div className="flex justify-between items-center mb-6">
            <div>
              <h1 className="text-2xl font-bold tracking-wide">Macro<span className="neon-text">Sense</span></h1>
              <p className="text-xs text-gray-400">AI-Powered Calorie & Macro Tracker</p>
            </div>
            <div className="text-right">
              <span className={`text-xs px-2 py-1 rounded ${geminiApiKey ? 'bg-emerald-950 text-emerald-400 border border-emerald-800' : 'bg-amber-950 text-amber-400 border border-amber-800'}`}>
                {geminiApiKey ? 'AI Active' : 'No API Key'}
              </span>
            </div>
          </div>

          {/* Tab Navigation */}
          <div className="flex bg-[#16191e] rounded-xl p-1 mb-6 border border-[#23272f]">
            <button 
              onClick={() => setActiveTab('dashboard')} 
              className={`flex-1 py-2 text-sm font-medium rounded-lg transition ${activeTab === 'dashboard' ? 'neon-bg text-black font-semibold' : 'text-gray-400'}`}
            >
              Dashboard
            </button>
            <button 
              onClick={() => setActiveTab('settings')} 
              className={`flex-1 py-2 text-sm font-medium rounded-lg transition ${activeTab === 'settings' ? 'neon-bg text-black font-semibold' : 'text-gray-400'}`}
            >
              Settings & AI
            </button>
          </div>

          {/* DASHBOARD TAB */}
          {activeTab === 'dashboard' && (
            <div className="space-y-4">
              {/* Daily Progress Overview */}
              <div className="card-dark p-5 rounded-2xl">
                <div className="flex justify-between items-baseline mb-2">
                  <span className="text-sm font-medium text-gray-400">Daily Energy</span>
                  <span className="text-xs text-gray-500">Target: {caloriesGoal} kcal</span>
                </div>
                <div className="text-3xl font-extrabold text-white mb-4">0 <span className="text-sm font-normal text-gray-400">/ {caloriesGoal} kcal</span></div>
                
                {/* Progress bar */}
                <div className="w-full bg-gray-800 h-2.5 rounded-full overflow-hidden mb-6">
                  <div className="neon-bg h-full w-0 transition-all duration-300"></div>
                </div>

                {/* Macro Distribution Grid */}
                <div className="grid grid-cols-3 gap-2 text-center">
                  <div className="bg-[#0d0f12] p-3 rounded-xl border border-gray-800">
                    <div className="text-xs text-gray-400">Protein</div>
                    <div className="text-lg font-bold text-emerald-400">0g</div>
                    <div className="text-[10px] text-gray-500">/ {proteinGoal}g</div>
                  </div>
                  <div className="bg-[#0d0f12] p-3 rounded-xl border border-gray-800">
                    <div className="text-xs text-gray-400">Carbs</div>
                    <div className="text-lg font-bold text-sky-400">0g</div>
                    <div className="text-[10px] text-gray-500">/ {carbsGoal}g</div>
                  </div>
                  <div className="bg-[#0d0f12] p-3 rounded-xl border border-gray-800">
                    <div className="text-xs text-gray-400">Fat</div>
                    <div className="text-lg font-bold text-amber-400">0g</div>
                    <div className="text-[10px] text-gray-500">/ {fatGoal}g</div>
                  </div>
                </div>
              </div>

              {/* Meals Quick Log */}
              <div className="space-y-3">
                {['Breakfast', 'Lunch', 'Dinner', 'Snacks'].map((meal) => (
                  <div key={meal} className="card-dark p-4 rounded-xl flex justify-between items-center">
                    <div>
                      <h3 className="font-semibold text-sm">{meal}</h3>
                      <p className="text-xs text-gray-500">0 kcal logged</p>
                    </div>
                    <button className="neon-bg text-black text-xs font-bold px-3 py-1.5 rounded-lg">+ Log</button>
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* SETTINGS & AI CONFIG TAB */}
          {activeTab === 'settings' && (
            <div className="space-y-6">
              {/* Gemini API Key Card */}
              <div className="card-dark p-5 rounded-2xl border-emerald-900/50">
                <div className="flex items-center gap-2 mb-2">
                  <span className="neon-text text-lg">â¨</span>
                  <h2 className="font-bold text-base text-white">Gemini AI Configuration</h2>
                </div>
                <p className="text-xs text-gray-400 mb-4">
                  Enter your Google Gemini API key to enable AI meal photo scanning and unlisted food macro estimation. Saved locally on your device.
                </p>

                <div className="space-y-3">
                  <div>
                    <label className="block text-xs font-medium text-gray-300 mb-1">Gemini API Key</label>
                    <input 
                      type="password"
                      placeholder="AIzaSy..."
                      value={geminiApiKey}
                      onChange={(e) => setGeminiApiKey(e.target.value)}
                      className="w-full bg-[#0d0f12] border border-gray-700 rounded-xl px-3 py-2.5 text-sm text-white focus:outline-none focus:border-emerald-500"
                    />
                  </div>

                  <button 
                    onClick={saveSettings}
                    className="w-full neon-bg text-black font-bold text-sm py-2.5 rounded-xl hover:opacity-90 transition"
                  >
                    Save Key & Preferences
                  </button>

                  {keySavedMessage && (
                    <div className="text-xs text-emerald-400 text-center font-medium pt-1">
                      â Settings & API key saved to local storage!
                    </div>
                  )}
                </div>
              </div>

              {/* Goals & Auto Balancer */}
              <div className="card-dark p-5 rounded-2xl">
                <h2 className="font-bold text-base text-white mb-4">Targets & Auto-Balancer</h2>

                <div className="space-y-4">
                  <div>
                    <label className="block text-xs text-gray-400 mb-1">Daily Calorie Goal (kcal)</label>
                    <input 
                      type="number"
                      value={caloriesGoal}
                      onChange={(e) => setCaloriesGoal(Number(e.target.value))}
                      className="w-full bg-[#0d0f12] border border-gray-800 rounded-xl px-3 py-2 text-sm text-white"
                    />
                  </div>

                  <div>
                    <label className="block text-xs text-gray-400 mb-1">Protein Goal (g) â Auto Balances Carbs/Fat</label>
                    <input 
                      type="number"
                      value={proteinGoal}
                      onChange={(e) => handleProteinChange(e.target.value)}
                      className="w-full bg-[#0d0f12] border border-gray-800 rounded-xl px-3 py-2 text-sm text-white"
                    />
                  </div>

                  <div className="grid grid-cols-2 gap-3">
                    <div>
                      <label className="block text-xs text-gray-400 mb-1">Carbs (g)</label>
                      <input 
                        type="number"
                        value={carbsGoal}
                        onChange={(e) => setCarbsGoal(Number(e.target.value))}
                        className="w-full bg-[#0d0f12] border border-gray-800 rounded-xl px-3 py-2 text-sm text-white"
                      />
                    </div>
                    <div>
                      <label className="block text-xs text-gray-400 mb-1">Fat (g)</label>
                      <input 
                        type="number"
                        value={fatGoal}
                        onChange={(e) => setFatGoal(Number(e.target.value))}
                        className="w-full bg-[#0d0f12] border border-gray-800 rounded-xl px-3 py-2 text-sm text-white"
                      />
                    </div>
                  </div>
                </div>
              </div>
            </div>
          )}
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById('root')).render(<App />);
  </script>
</body>
</html>
